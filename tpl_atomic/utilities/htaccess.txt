<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "geolocation=(), camera=(), microphone=(), payment=(), interest-cohort=()"

  # Uncomment HSTS only after confirming the full site (and subdomains you care
  # about) are served exclusively over HTTPS. Drop `preload` if you don't plan
  # to submit to https://hstspreload.org/ — once preloaded, removal is slow.
  # Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"

  # CSP starter — Report-Only is safe to enable by default; it flags violations
  # in the browser console without breaking anything. Tighten the policy and
  # switch to Content-Security-Policy (enforced) once you've reviewed reports.
  Header always set Content-Security-Policy-Report-Only "default-src 'self' https: 'unsafe-inline' 'unsafe-eval'; frame-ancestors 'self'; object-src 'none'; base-uri 'self';"

  # ----------------------------------------------------------------
  # OPT-IN: a concrete policy matched to Atomic's actual external surface.
  # Everything below is deliberately commented out — nothing here is ever
  # applied automatically. Review it, trim it to the sources your template
  # settings actually use, then enable ONE of the two headers (and remove
  # the generic Report-Only header above so the two don't overlap).
  #
  # CDN origins the Atomic template settings can load from:
  #   cdn.jsdelivr.net          Bootstrap/Bootswatch CSS+JS, Bootstrap Icons
  #                             (bootstrapsource=2 or 6-14, loadbsicons=1)
  #   cdnjs.cloudflare.com      Font Awesome CSS or JS (fontawesome=2 or 3)
  #   fonts.googleapis.com      Google Fonts stylesheets (headerfont/bodyfont
  #                             values 1 and 3-12)
  #   fonts.gstatic.com         Google Fonts font files
  #   code.jquery.com           jQuery from CDN (jqlibrary=1 or 2)
  #   unpkg.com                 ScrollReveal (scrollreveal=1)
  #   www.googletagmanager.com  GA4 loader gtag.js (gacode set)
  #   *.google-analytics.com    GA4 measurement hits, including the regional
  #   *.analytics.google.com    endpoints (region1.google-analytics.com etc.)
  #
  # THE INLINE-SCRIPT/STYLE STORY — read this before enforcing anything.
  # Atomic itself emits inline code on every page:
  #   - the pre-paint theme-resolution IIFE and the `var defaultTheme`
  #     snippet are inline <script> blocks in the <head>;
  #   - the :root { --atomic-* } design-token block is an inline <style>;
  #   - the lazy-stylesheet trick onload="this.media='all'" is an inline
  #     event handler, which CSP also treats as inline script.
  # A static .htaccess header cannot issue per-request nonces, so this
  # snippet takes the simplest honest route: 'unsafe-inline' in both
  # script-src and style-src. If you want a stricter nonce-based policy,
  # do NOT set CSP here — enable Joomla's core "System - HTTP Headers"
  # plugin instead, which generates nonces for inline assets it manages.
  #
  # Step 1 — enable the tailored policy in Report-Only mode and watch the
  # browser console / report endpoint for violations:
  # Header always set Content-Security-Policy-Report-Only "default-src 'self'; script-src 'self' 'unsafe-inline' cdn.jsdelivr.net cdnjs.cloudflare.com code.jquery.com unpkg.com www.googletagmanager.com; style-src 'self' 'unsafe-inline' cdn.jsdelivr.net cdnjs.cloudflare.com fonts.googleapis.com; font-src 'self' fonts.gstatic.com cdn.jsdelivr.net cdnjs.cloudflare.com; img-src 'self' data: www.googletagmanager.com *.google-analytics.com; connect-src 'self' *.google-analytics.com *.analytics.google.com www.googletagmanager.com; frame-ancestors 'self'; object-src 'none'; base-uri 'self';"
  #
  # Step 2 — once the console is clean, switch to the enforced header:
  # Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' cdn.jsdelivr.net cdnjs.cloudflare.com code.jquery.com unpkg.com www.googletagmanager.com; style-src 'self' 'unsafe-inline' cdn.jsdelivr.net cdnjs.cloudflare.com fonts.googleapis.com; font-src 'self' fonts.gstatic.com cdn.jsdelivr.net cdnjs.cloudflare.com; img-src 'self' data: www.googletagmanager.com *.google-analytics.com; connect-src 'self' *.google-analytics.com *.analytics.google.com www.googletagmanager.com; frame-ancestors 'self'; object-src 'none'; base-uri 'self';"
  #
  # Trim what you don't use: drop unpkg.com if ScrollReveal is off, drop
  # code.jquery.com if jQuery comes from Joomla or is disabled, drop the
  # Google entries if you use neither Google Fonts nor GA. If you filled in
  # any of the Custom snippet fields (Bootstrap, jQuery, Font Awesome,
  # Google Fonts, Custom Code), add those hosts to the matching directives.
  # Extensions and embedded content (maps, videos, form services) may need
  # further origins — that is what the Report-Only step is for.
  # ----------------------------------------------------------------

  # Hide server/PHP fingerprints from response headers
  Header unset X-Powered-By
  Header unset Server
</IfModule>

RedirectMatch 404 /\.hg(/|$)

RedirectMatch 404 /yarn\.lock$
RedirectMatch 404 /Gemfile(\.lock)?$
RedirectMatch 404 /phpunit(\.xml(\.dist)?)?$

# Block cloud-credential, SSH, and host-config file probes
RedirectMatch 404 (?i)/\.(aws|npmrc|htpasswd|netrc|dockerenv|kube|ssh)(/|$)
RedirectMatch 404 (?i)/(phpinfo|php|info|i|test)\.php$
RedirectMatch 404 (?i)/wp-config(-(old|backup|sample))?\.php

# Block PHP files under /.well-known/ (common backdoor-planting spot, e.g. nastar.php)
RedirectMatch 403 ^/\.well-known/.*\.php$

# Block probes for temp, backup, and staging directories
RedirectMatch 403 ^/(backup|backups|bak|old|db|database|dump|dumps|sql|mysql|temp|tmp|test|tests|dev|staging|stage|new|beta)/?$

# Block known recon/exploit paths from scanners-in-the-wild (return 410 Gone
# to discourage retries):
#   /Readme.txt           — Chinese scanner CMS-fingerprint pull
#   /dayrui/              — dayrui/Fcms scanner fingerprint
#   /assets/plugins/elfinder — CVE-2021-32682 elFinder RCE probe
#   /administrator/manifests — Joomla manifest-leak probe
#   /phpmyadmin           — universal admin-panel probe
RedirectMatch 410 (?i)^/(Readme\.txt|dayrui|phpmyadmin|administrator/manifests|assets/plugins/elfinder)

# Block direct access to Joomla sensitive files. configuration.php is the
# Joomla equivalent of wp-config.php (database credentials live here).
<FilesMatch "^(configuration\.php|\.htaccess|\.htpasswd|web\.config|htaccess\.txt|\.user\.ini|.*\.(bak|sql|swp|swo|log|ini|conf|old|orig|save|tar|tar\.gz|zip|rar))$">
  Require all denied
</FilesMatch>

# Block external access to Joomla's cron.php (CLI/cron should run via PHP CLI,
# not the web). Comment out if you've configured webcron via an external service.
<Files "cron.php">
  Require all denied
</Files>

<IfModule mod_rewrite.c>
  RewriteEngine On

  # ---- HTTPS enforcement ----
  # Uncomment once you've confirmed a valid TLS certificate covers every
  # hostname this site answers to. Leaving HSTS off (above) AND this rule off
  # is fine; turning HSTS on while this is off is a footgun.
  # RewriteCond %{HTTPS} !=on
  # RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]

  # ---- Bot early-block ----
  # Block bots flagged via the bad_bot env (set in Section 3 below) BEFORE
  # any redirect runs. Apache's order/deny + RequireAll run at the
  # authorization phase, which is AFTER mod_rewrite's URL-translation phase,
  # so a canonical-host 301 (e.g. non-www → www) would otherwise fire before
  # the bot is denied. This guard short-circuits at the rewrite phase.
  RewriteCond %{ENV:bad_bot} =1
  RewriteRule .* - [F,L]

  # ---- Referrer-spam / log-poisoning filter ----
  # Spoofed referrers like www.google.net, www.yandex.net, binance.com show
  # up in awstats/analytics to bait clicks. Block on sight.
  RewriteCond %{HTTP_REFERER} (google|yandex|baidu|yahoo|bing)\.net [NC,OR]
  RewriteCond %{HTTP_REFERER} (semalt|kambasoft|savetubevideo|buttons-for-website)\.com [NC,OR]
  RewriteCond %{HTTP_REFERER} (dayrui|Readme\.txt|usingjoomlabook\.com) [NC]
  RewriteRule .* - [F,L]

  # ---- WordPress author enumeration ----
  # Scanners spray ?author=N indiscriminately; pointless on Joomla.
  RewriteCond %{QUERY_STRING} (^|&)author=\d+ [NC]
  RewriteRule .* - [F,L]

  # ---- Targeted Joomla Astroid installTemplate exploit ----
  RewriteCond %{QUERY_STRING} astroid=installTemplate [NC]
  RewriteRule .* - [F,L]

  # ---- Common SQLi / RFI / LFI / RCE patterns in query strings ----
  RewriteCond %{QUERY_STRING} (\.\./){2,}            [NC,OR]
  RewriteCond %{QUERY_STRING} (union[\s+]+select)    [NC,OR]
  RewriteCond %{QUERY_STRING} (concat[\s+]*\()       [NC,OR]
  RewriteCond %{QUERY_STRING} (benchmark[\s+]*\()    [NC,OR]
  RewriteCond %{QUERY_STRING} (sleep[\s+]*\([\s+]*[0-9]) [NC,OR]
  RewriteCond %{QUERY_STRING} (passthru|shell_exec|system|phpinfo)[\s+]*\( [NC,OR]
  RewriteCond %{QUERY_STRING} (\bfile_get_contents\b|\bfopen\b)\( [NC,OR]
  RewriteCond %{QUERY_STRING} (=|%3D)(php|https?|ftp|data|file)(:|%3A)// [NC,OR]
  RewriteCond %{QUERY_STRING} GLOBALS(=|\[|\%[0-9A-Z]{0,2}) [OR]
  RewriteCond %{QUERY_STRING} _REQUEST(=|\[|\%[0-9A-Z]{0,2}) [OR]
  RewriteCond %{QUERY_STRING} base64_(en|de)code[^(]*\([^)]*\) [NC]
  RewriteRule .* - [F,L]

  # ---- No-UA requests against dynamic endpoints ----
  # Legitimate browsers always send a User-Agent. An empty UA hitting a
  # PHP/CGI/Perl script is almost always a scripted exploit attempt.
  RewriteCond %{HTTP_USER_AGENT} ^-?$
  RewriteCond %{REQUEST_URI}     \.(php|cgi|pl)$ [NC]
  RewriteRule .* - [F,L]
</IfModule>

  Header set Cache-Control "max-age=2628000, public"
</filesMatch>

  Header set Cache-Control "max-age=31536000, public"

</filesMatch>
